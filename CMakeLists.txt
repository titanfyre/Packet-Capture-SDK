cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(PacketCaptureSDK VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Build on Windows x64 with MSVC.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "Instrument user-mode targets" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /EHsc /utf-8 /permissive- /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(PacketCapture SDK/PacketCapture.cpp)
target_link_libraries(PacketCapture PUBLIC winhttp crypt32)
add_executable(PacketCaptureStudio WIN32 App/Main.cpp)
target_link_libraries(PacketCaptureStudio PRIVATE PacketCapture user32 gdi32 comctl32)
target_link_options(PacketCaptureStudio PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_executable(PacketCaptureCLI App/Console.cpp)
add_executable(PacketCaptureExample Examples/frames.cpp)
add_executable(PacketCaptureTests Tests/Protocol.cpp)
foreach(t PacketCaptureCLI PacketCaptureExample PacketCaptureTests)
 target_link_libraries(${t} PRIVATE PacketCapture)
endforeach()
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Exact opaque signed TitanPacketCapture package")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 foreach(n TitanPacketCapture.inf TitanPacketCapture.cat TitanPacketCapture.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${n}")
   message(FATAL_ERROR "Incomplete driver package: ${n}")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanPacketCapture.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanPacketCapture.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanPacketCapture.sys\"\n")
endif()
add_library(PacketSetupCore Installer/Setup.cpp)
target_link_libraries(PacketSetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32 uuid)
add_executable(PacketCaptureSetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
add_executable(PacketCaptureSetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
foreach(t PacketCaptureSetup PacketCaptureSetupConsole)
 target_link_libraries(${t} PRIVATE PacketSetupCore user32 gdi32)
 target_link_options(${t} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
foreach(t PacketCaptureStudio PacketCaptureCLI PacketCaptureExample PacketCaptureTests PacketCaptureSetup PacketCaptureSetupConsole)
 set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
enable_testing()
add_test(NAME FrameBoundsAndParsing COMMAND PacketCaptureTests)

add_executable(PacketInstallerTests Tests/Installer.cpp)
target_link_libraries(PacketInstallerTests PRIVATE PacketSetupCore)
set_target_properties(PacketInstallerTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME InstallerIdentityAndUnsignedRefusal COMMAND PacketInstallerTests)
